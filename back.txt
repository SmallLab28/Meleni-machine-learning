#define SPACE_TOKEN "\xc4\xa0"


// Xác định ký tự đầu tiên chiếm bao nhiêu bytes trong UTF-8
size_t get_utf8_len(unsigned char first_byte) {
    if ((first_byte & 0x80) == 0)    return 1; // ASCII thường (1 byte)
    if ((first_byte & 0xE0) == 0xC0) return 2; // UTF-8 2 bytes (như \xc4\xa0 - chữ Ġ)
    if ((first_byte & 0xF0) == 0xE0) return 3; // UTF-8 3 bytes (tiếng Việt: à, á...)
    if ((first_byte & 0xF8) == 0xF0) return 4; // UTF-8 4 bytes (Emoji...)
    return 1;
}


std::string replace_spaces(const std::string &content) {
    std::string new_content;
    for (size_t i = 0; i < content.size(); i++) {
        if (content[i] == ' ') {
            new_content += SPACE_TOKEN; // Ký tự 'Ġ' trong UTF-8
        } else {
            new_content += content[i];
        }
    }
    return new_content;
}

std::vector<std::string> split_utf8(const std::string &str) {
    std::vector<std::string> chars;
    size_t i = 0;
    while (i < str.size()) {
        unsigned char c = (unsigned char)str[i];
        int char_len = 1;
        
        if ((c & 0x80) == 0)         char_len = 1; 
        else if ((c & 0xE0) == 0xC0) char_len = 2; 
        else if ((c & 0xF0) == 0xE0) char_len = 3;
        else if ((c & 0xF8) == 0xF0) char_len = 4;
        chars.push_back(str.substr(i, char_len));
        i += char_len;
    }
    return chars;
}
